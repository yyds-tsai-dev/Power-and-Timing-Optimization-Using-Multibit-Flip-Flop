    box/.style={draw, rounded corners=1pt, inner sep=2.5pt, align=center},

  ]
  % ---------------- left: committed state (read-only) ----------------
  \node[box, text width=24mm] (caches) at (1.50,4.80)
    {immutable caches\\ (per logical FF/gate)};

    {live positions +\\ cell bindings\\ (logical$\to$physical\\ indirection)};
  \node[draw, dashed, gray!70, rounded corners=1pt, inner sep=1.5mm,

  \node[note, font=\footnotesize\bfseries, anchor=south, inner sep=2pt]
    at (committed.north) {committed state (read-only)};

  % ---------------- left, below: candidate move as override map -------
  \node[ovr, text width=25mm] (ovr) at (1.50,1.55)
    {candidate structural move:\\ override map\\ bit $\to$ (D, Q, $t_{cq}$)};

  \node[note, font=\footnotesize\bfseries] at (6.00,5.30) {forward cone walk};
  \node[note, gray, text width=44mm] at (6.00,4.85)
    {override-first resolution;\\ thread-local scratch, zero writes};

  \fill[blue!8] (3.75,3.35) -- (5.20,4.25) -- (7.60,4.25)
                -- (7.60,2.45) -- (5.20,2.45) -- cycle;

  \node[gate] (g1)  at (5.55,3.95) {g$_1$};
  \node[gate] (g2)  at (5.55,2.75) {g$_2$};
  \node[cell] (ffs1) at (7.10,3.95) {FF$_{s1}$};
  \node[cell] (ffs2) at (7.10,2.75) {FF$_{s2}$};
  \draw[wire] (ffm.east) -- (g1.west);
  \draw[wire] (ffm.east) -- (g2.west);
  \draw[wire] (g1.east)  -- (ffs1.west);
  \draw[wire] (g2.east)  -- (ffs2.west);

  % ---------------- flow arrows into the walk -------------------------
  \draw[wire] (committed.east) -- (ffm.west)
    node[note, gray, pos=0.72, above=1.6mm] {read};

  \node[box] (delta) at (4.35,0.65)

  \node[note, text=red!70!black, anchor=west, align=left] (rej) at (6.30,0.20)

  \draw[wire] (delta.east) -- (acc.west);
  \draw[wire] (delta.east) -- (rej.west);
